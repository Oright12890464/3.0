function esc(v){
  return String(v??'').replace(/[&<>"']/g,c=>({
    '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
  }[c]));
}
function setBusy(v){
  busy=v;
  $('app').classList.toggle('busy',v);
  $('searchBtn').disabled=v;
  $('submitBtn').disabled=v;
}
function showStatus(msg,type='ok'){
  $('status').textContent=msg;
  $('status').className='status '+type;
  window.scrollTo({top:0,behavior:'smooth'});
}

  $('status').textContent='';
  $('status').className='status';

function search(){
  if(busy)return;
  hideStatus();
  $('results').innerHTML='<div class="empty">查詢中…</div>';
  $('count').textContent='正在搜尋八大倉的品號與儲位…';
  setBusy(true);
  google.script.run
    .withSuccessHandler(data=>{
      setBusy(false);
      currentResults=data.results||[];
      renderResults(currentResults);
      if(currentDetailIndex!==null&&currentResults[currentDetailIndex])openDetail(currentDetailIndex);
    })
    .withFailureHandler(err=>{
      setBusy(false);
      $('results').innerHTML='';
      $('count').textContent='查詢失敗';
      showStatus(err.message||String(err),'err');
    })
    .searchMaterials($('query').value.trim());
}

/**
 * 只刷新目前操作的那一筆物料
 */
function refreshOneRecord(sheetName, rowNumber, message) {
  google.script.run
    .withSuccessHandler(item => {
      const index = currentResults.findIndex(
        x => x.sheetName === sheetName && x.row === rowNumber
      );

      if (index >= 0) {
        currentResults[index] = item;

        // 更新目前詳細資料畫面
        if (
          currentDetailIndex === index &&
          !$('detailView').classList.contains('hidden')
        ) {
          editMode = false;
          $('detailHeading').textContent =
            item.sheetName + '｜' +
            (item.itemNo || '無品號') + '｜' +
            (item.itemName || '無品名');

          renderDetailFields(item);
          $('detailActions').innerHTML = defaultDetailActions();
          bindDetailActions();
        }

        // 同步更新查詢清單中這筆資料
        renderResults(currentResults);
      }

      if (message) showStatus(message, 'ok');
    })
    .withFailureHandler(err => {
      showStatus(
        '異動已完成，但刷新資料失敗：' + (err.message || String(err)),
        'err'
      );
    })
    .getMaterialRecord(sheetName, rowNumber);
}
function renderResults(items){
  $('count').textContent='共 '+items.length+' 筆資料';
  if(!items.length){
    $('results').innerHTML='<div class="empty">沒有找到符合的資料，請換一個關鍵字。</div>';
    return;
  }
  $('results').innerHTML=items.map((x,i)=>{
    return '<article class="item" data-index="'+i+'" tabindex="0" role="button" aria-label="查看 '+esc(x.itemName||x.itemNo||'物料')+' 詳細資料">'+
      '<div class="warehouse"><strong>'+esc(x.sheetName)+'</strong><span>倉別</span></div>'+
      '<div class="details"><div class="name">'+esc(x.itemName||'（未填品名）')+'</div>'+
      '<div class="meta"><span>品號：'+esc(x.itemNo||'—')+'</span><span>批號：'+esc(x.batch||'—')+'</span></div>'+
      '<div class="hint">點擊查看完整資料　›</div></div></article>';
  }).join('');
}
function openDetail(index){
  const item=currentResults[index];
  if(!item)return;
  currentDetailIndex=index;
  editMode=false;
  $('listView').classList.add('hidden');
  $('detailView').classList.remove('hidden');
  $('detailHeading').textContent=item.sheetName+'｜'+(item.itemNo||'無品號')+'｜'+(item.itemName||'無品名');
  renderDetailFields(item);
  $('detailActions').innerHTML=defaultDetailActions();
  bindDetailActions();
  window.scrollTo({top:0,behavior:'smooth'});
}
function normalizeLabel(label){
  return String(label||'').replace(/[\\s\\r\\n：:（）()]/g,'').toLowerCase();
}
function isProtectedField(label,col,item){
  const n=normalizeLabel(label);
  const protectedNames=['品號','品名','數量1','數量2','數量','erp','確認','單位數量','異動','差異'];
  if(protectedNames.includes(n))return true;
  const protectedCols=[
    item.itemNoCol,item.itemNameCol,item.qty1Col,item.qty2Col,
    item.erpCol,item.confirmCol,item.unitQtyCol,item.movementCol,item.diffCol
  ].filter(Boolean);
  return protectedCols.includes(col);
}
function renderDetailFields(item){
  const fields=Array.isArray(item.allFields)?item.allFields:[];
  const fallback=['進料日期','批號','有效日期','品號','品名','ERP','庫位1','數量1','庫位2','數量2','差異','前置包裝','備註','確認','箱/板','單位數量','異動'];
  $('detailFields').innerHTML=fields.map((v,i)=>{
    const label=(item.headers&&item.headers[i])?item.headers[i]:fallback[i]||('欄位'+(i+1));
    const col=i+1,locked=isProtectedField(label,col,item),value=v===''?'':String(v);
    let content=esc(value||'—');
    if(editMode&&!locked){
      content='<input class="detail-edit-input" data-col="'+col+'" value="'+esc(value)+'" placeholder="（空白）" aria-label="編輯'+esc(label)+'">';
    }
    return '<div class="detail-row"><div class="detail-label">'+esc(label)+(editMode&&!locked?' <span style="color:var(--accent)">✎</span>':'')+'</div><div class="detail-value">'+content+'</div></div>';
  }).join('');
}
function defaultDetailActions(){

}

  $('detailEdit').addEventListener('click',toggleEditMode);
}
function toggleEditMode(){
  if(currentDetailIndex===null)return;
  editMode=true;
  renderDetailFields(currentResults[currentDetailIndex]);
  $('detailActions').innerHTML='<button class="primary" id="saveEdit">儲存修改</button><button id="cancelEdit">取消編輯</button>';
  $('saveEdit').addEventListener('click',saveDetailEdits);
  $('cancelEdit').addEventListener('click',()=>{
    editMode=false;
    renderDetailFields(currentResults[currentDetailIndex]);
    $('detailActions').innerHTML=defaultDetailActions();
    bindDetailActions();
  });
}
function saveDetailEdits() {
  if (busy || currentDetailIndex === null) return;

  const item = currentResults[currentDetailIndex];

  const fields = Array.from(
    document.querySelectorAll('.detail-edit-input')
  ).map(el => ({
    col: Number(el.dataset.col),
    value: el.value
  }));

  const payload = {
    sheetName: item.sheetName,
    row: item.row,
    fields: fields
  };

  setBusy(true);

  google.script.run
    .withSuccessHandler(result => {
      setBusy(false);

      refreshOneRecord(
        item.sheetName,
        item.row,
        result.message || '資料修改完成。'
      );
    })
    .withFailureHandler(err => {
      setBusy(false);
      showStatus(err.message || String(err), 'err');
    })
    .updateMaterialFields(payload);
}

    return;
  }
  const item=currentResults[index];
  if(!item)return;
  currentOperation={item,kind,action};
  const double=item.type==='double',isIn=action==='in';
  $('modalTitle').textContent=kind==='location'?'修改庫位':(isIn?'入庫':'出庫');
  $('modalDesc').textContent=item.sheetName+'｜'+(item.itemNo||'無品號')+'｜'+(item.itemName||'無品名')+
    (item.batch?'｜批號 '+item.batch:'')+
    (kind==='movement'?'。只會異動有填寫數量的欄位，留空的欄位不會變動。':'。庫位1與庫位2分開修改，留空欄位不會變動。');
  let body='';
  if(kind==='movement'){
    body+=slotHtml(double?'儲位1／數量1':'庫位／數量',item.loc1,item.qty1,'qty1',true);
    if(double)body+=slotHtml('儲位2／數量2',item.loc2,item.qty2,'qty2',true);
    body+='<div class="field"><label for="opNote">異動備註（選填）</label><textarea id="opNote" rows="2" placeholder="例如：產線退料、領料單號"></textarea></div>';
  }else{
    body+=slotHtml(double?'儲位1／庫位1':'庫位',item.loc1,item.qty1,'loc1',false);
    if(double)body+=slotHtml('儲位2／庫位2',item.loc2,item.qty2,'loc2',false);
    body+='<div class="hint">只會修改有填寫的新庫位；未填寫的儲位保持原樣。</div>';
  }
  $('modalBody').innerHTML=body;
  $('submitBtn').textContent=kind==='location'?'儲存庫位':'確認'+(isIn?'入庫':'出庫');
  $('overlay').classList.add('show');
}
function slotHtml(title,loc,qty,id,isQuantity){
  const hint=isQuantity?'目前庫位：'+(loc||'未設定')+'｜目前數量：'+(qty||'0'):'目前庫位：'+(loc||'未設定');
  const label=isQuantity?'輸入'+(currentOperation.action==='in'?'入庫':'出庫')+'數量':'輸入新庫位';
  return '<div class="slot"><div class="slot-title">'+esc(title)+'</div><div class="hint">'+esc(hint)+'</div><div class="field">'+
    '<label for="'+id+'">'+label+'</label><input id="'+id+'" '+(isQuantity?'type="number" min="0.000001" step="any"':'type="text"')+
    ' placeholder="'+(isQuantity?'留空則不異動':'留空代表不修改')+'"></div></div>';
}
function submitModal() {
  if (busy || !currentOperation) return;

  const op = currentOperation;
  const item = op.item;

  const payload = {
    sheetName: item.sheetName,
    row: item.row
  };

  if (op.kind === 'movement') {
    payload.action = op.action;
    payload.operator = $('operatorSelect').value;
    payload.qty1 = $('qty1') ? $('qty1').value : '';
    payload.qty2 = $('qty2') ? $('qty2').value : '';
    payload.note = $('opNote') ? $('opNote').value : '';

    if (payload.qty1 === '' && payload.qty2 === '') {
      showStatus('請至少輸入一個數量；留空欄位不會異動。', 'err');
      return;
    }
  } else {
    payload.loc1 = $('loc1') ? $('loc1').value : '';
    payload.loc2 = $('loc2') ? $('loc2').value : '';

    if (!payload.loc1.trim() && !payload.loc2.trim()) {
      showStatus('請至少填寫一個要修改的庫位。', 'err');
      return;
    }
  }

  setBusy(true);
  $('submitBtn').textContent = '處理中…';

  const runner = google.script.run
    .withSuccessHandler(result => {
      setBusy(false);
      closeModal();

      refreshOneRecord(
        item.sheetName,
        item.row,
        result.message || '操作完成。'
      );
    })
    .withFailureHandler(err => {
      setBusy(false);
      $('submitBtn').textContent = '確認執行';
      showStatus(err.message || String(err), 'err');
    });

  if (op.kind === 'movement') {
    runner.performStockMovement(payload);
  } else {
    runner.updateMaterialLocations(payload);
  }
}
function closeModal(){
  $('overlay').classList.remove('show');
  currentOperation=null;
}
