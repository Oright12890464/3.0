function esc(v){
  return String(v??'').replace(/[&<>"']/g,c=>({
    '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
  }[c]));
}
function setBusy(v){
  busy=v;
  $('app').classList.toggle('busy',v);
  $('searchBtn').disabled=v;
  $('submitBtn').disabled=v;
}
function showStatus(msg,type='ok'){
  $('status').textContent=msg;
  $('status').className='status '+type;
  window.scrollTo({top:0,behavior:'smooth'});
}

  $('status').textContent='';
  $('status').className='status';

function search(){
  if(busy)return;
  hideStatus();
  $('results').innerHTML='<div class="empty">查詢中…</div>';
  $('count').textContent='正在搜尋八大倉的品號與儲位…';
  setBusy(true);
  google.script.run
    .withSuccessHandler(data=>{
      setBusy(false);
      currentResults=data.results||[];
      renderResults(currentResults);
      if(currentDetailIndex!==null&&currentResults[currentDetailIndex])openDetail(currentDetailIndex);
    })
    .withFailureHandler(err=>{
      setBusy(false);
      $('results').innerHTML='';
      $('count').textContent='查詢失敗';
      showStatus(err.message||String(err),'err');
    })
    .searchMaterials($('query').value.trim());
}

/**
 * 只刷新目前操作的那一筆物料
 */
function refreshOneRecord(sheetName, rowNumber, message) {
  google.script.run
    .withSuccessHandler(item => {
      const index = currentResults.findIndex(
        x => x.sheetName === sheetName && x.row === rowNumber
      );

      if (index >= 0) {
        currentResults[index] = item;

        // 更新目前詳細資料畫面
        if (
          currentDetailIndex === index &&
          !$('detailView').classList.contains('hidden')
        ) {
          editMode = false;
          $('detailHeading').textContent =
            item.sheetName + '｜' +
            (item.itemNo || '無品號') + '｜' +
            (item.itemName || '無品名');

          renderDetailFields(item);
          $('detailActions').innerHTML = defaultDetailActions();
          bindDetailActions();
        }

        // 同步更新查詢清單中這筆資料
        renderResults(currentResults);
      }

      if (message) showStatus(message, 'ok');
    })
    .withFailureHandler(err => {
      showStatus(
        '異動已完成，但刷新資料失敗：' + (err.message || String(err)),
        'err'
      );
    })
    .getMaterialRecord(sheetName, rowNumber);
}
function renderResults(items){
  $('count').textContent='共 '+items.length+' 筆資料';
  if(!items.length){
    $('results').innerHTML='<div class="empty">沒有找到符合的資料，請換一個關鍵字。</div>';
    return;
  }
  $('results').innerHTML=items.map((x,i)=>{
    return '<article class="item" data-index="'+i+'" tabindex="0" role="button" aria-label="查看 '+esc(x.itemName||x.itemNo||'物料')+' 詳細資料">'+
      '<div class="warehouse"><strong>'+esc(x.sheetName)+'</strong><span>倉別</span></div>'+
      '<div class="details"><div class="name">'+esc(x.itemName||'（未填品名）')+'</div>'+
      '<div class="meta"><span>品號：'+esc(x.itemNo||'—')+'</span><span>批號：'+esc(x.batch||'—')+'</span></div>'+
      '<div class="hint">點擊查看完整資料　›</div></div></article>';
  }).join('');
}
function openDetail(index){
  const item=currentResults[index];
  if(!item)return;
  currentDetailIndex=index;
  editMode=false;
  $('listView').classList.add('hidden');
  $('detailView').classList.remove('hidden');
  $('detailHeading').textContent=item.sheetName+'｜'+(item.itemNo||'無品號')+'｜'+(item.itemName||'無品名');
  renderDetailFields(item);
  $('detailActions').innerHTML=defaultDetailActions();
  bindDetailActions();
  window.scrollTo({top:0,behavior:'smooth'});
}
function normalizeLabel(label){
  return String(label||'').replace(/[\\s\\r\\n：:（）()]/g,'').toLowerCase();
}
function isProtectedField(label,col,item){
  const n=normalizeLabel(label);
  const protectedNames=['品號','品名','數量1','數量2','數量','erp','確認','單位數量','異動','差異'];
  if(protectedNames.includes(n))return true;
  const protectedCols=[
    item.itemNoCol,item.itemNameCol,item.qty1Col,item.qty2Col,
    item.erpCol,item.confirmCol,item.unitQtyCol,item.movementCol,item.diffCol
  ].filter(Boolean);
  return protectedCols.includes(col);
}
function renderDetailFields(item){
  const fields=Array.isArray(item.allFields)?item.allFields:[];
  const fallback=['進料日期','批號','有效日期','品號','品名','ERP','庫位1','數量1','庫位2','數量2','差異','前置包裝','備註','確認','箱/板','單位數量','異動'];
  $('detailFields').innerHTML=fields.map((v,i)=>{
    const label=(item.headers&&item.headers[i])?item.headers[i]:fallback[i]||('欄位'+(i+1));
    const col=i+1,locked=isProtectedField(label,col,item),value=v===''?'':String(v);
    let content=esc(value||'—');
    if(editMode&&!locked){
      content='<input class="detail-edit-input" data-col="'+col+'" value="'+esc(value)+'" placeholder="（空白）" aria-label="編輯'+esc(label)+'">';
    }
    return '<div class="detail-row"><div class="detail-label">'+esc(label)+(editMode&&!locked?' <span style="color:var(--accent)">✎</span>':'')+'</div><div class="detail-value">'+content+'</div></div>';
  }).join('');
}
function defaultDetailActions(){

}

  $('detailEdit').addEventListener('click',toggleEditMode);
}
function toggleEditMode(){
  if(currentDetailIndex===null)return;
  editMode=true;
  renderDetailFields(currentResults[currentDetailIndex]);
  $('detailActions').innerHTML='<button class="primary" id="saveEdit">儲存修改</button><button id="cancelEdit">取消編輯</button>';
  $('saveEdit').addEventListener('click',saveDetailEdits);
  $('cancelEdit').addEventListener('click',()=>{
    editMode=false;
    renderDetailFields(currentResults[currentDetailIndex]);
    $('detailActions').innerHTML=defaultDetailActions();
    bindDetailActions();
  });
}
function saveDetailEdits() {
  if (busy || currentDetailIndex === null) return;

  const item = currentResults[currentDetailIndex];

  const fields = Array.from(
    document.querySelectorAll('.detail-edit-input')
  ).map(el => ({
    col: Number(el.dataset.col),
    value: el.value
  }));

  const payload = {
    sheetName: item.sheetName,
    row: item.row,
    fields: fields
  };

  setBusy(true);

  google.script.run
    .withSuccessHandler(result => {
      setBusy(false);

      refreshOneRecord(
        item.sheetName,
        item.row,
        result.message || '資料修改完成。'
      );
    })
    .withFailureHandler(err => {
      setBusy(false);
      showStatus(err.message || String(err), 'err');
    })
    .updateMaterialFields(payload);
}

    return;
  }
  const item=currentResults[index];
  if(!item)return;
  currentOperation={item,kind,action};
  const double=item.type==='double',isIn=action==='in';
  $('modalTitle').textContent=kind==='location'?'修改庫位':(isIn?'入庫':'出庫');

/**
 * 將符合條件的資料列整理成前端需要的物料物件。
 * 搜尋與單筆刷新共用相同欄位格式，避免 makeRecord_ 未定義。
 */
function makeRecord_(c, rowNumber, shownRow, headers, dateSort) {
  return {
    sheetName: c.name,
    sheetOrder: c.order,
    type: c.type,
    row: rowNumber,
    itemNo: at_(shownRow, c.itemCol),
    itemName: at_(shownRow, c.nameCol),
    batch: at_(shownRow, c.batchCol),
    dateText: c.dateCol ? at_(shownRow, c.dateCol) : '',
    dateSort: dateSort || 0,
    loc1: at_(shownRow, c.loc1Col),
    qty1: at_(shownRow, c.qty1Col),
    loc2: c.qty2Col ? at_(shownRow, c.loc2Col) : '',
    qty2: c.qty2Col ? at_(shownRow, c.qty2Col) : '',
    headers: headers,
    allFields: shownRow.slice(0, c.cols)
  };
}

function searchMaterials(keyword) {
  // 只比對品號、儲位1、儲位2；不搜尋品名、批號、備註或其他欄位。
  const q = String(keyword || '').trim().toLowerCase();
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const out = [];

  SHEET_CONFIG.forEach(c => {
    const sh = ss.getSheetByName(c.name);
    if (!sh) return;

    const lastRow = sh.getLastRow();
    if (lastRow < 2) return;

    const numRows = lastRow - 1;

    // 每張表只讀一次顯示資料；搜尋時只檢查設定中的品號與儲位欄。
    // 避免原本同時讀取整張表的 getValues() + getDisplayValues()。
    const shown = sh.getRange(2, 1, numRows, c.cols).getDisplayValues();

    // 日期原始值只讀日期欄，不再把整張表的原始值全部讀一次。
    const dateValues = c.dateCol
      ? sh.getRange(2, c.dateCol, numRows, 1).getValues()
      : null;

    let headers = null;
    const matchedRows = [];

    shown.forEach((r, i) => {
      const item = at_(r, c.itemCol);
      const loc1 = at_(r, c.loc1Col);
      const loc2 = c.loc2Col ? at_(r, c.loc2Col) : '';

      // 空白資料列略過；儲位2不存在的倉別不會讀取或搜尋不存在的欄位。
      if (!item && !loc1 && !loc2) return;

      // 僅比對品號與實際存在的儲位欄位。
      if (q) {
        const matched = [item, loc1, loc2].some(value =>
          String(value || '').toLowerCase().includes(q)
        );
        if (!matched) return;
      }

      let dateSort = 0;
      if (dateValues) {
        const dateValue = dateValues[i][0];
        if (dateValue instanceof Date && !isNaN(dateValue.getTime())) {
          dateSort = dateValue.getTime();
        } else if (dateValue) {
          dateSort = Date.parse(dateValue) || 0;
        }
      }

      matchedRows.push({ rowNumber: i + 2, shownRow: r, dateSort: dateSort });
    });

    // 沒有符合的資料就不額外讀取標題列。
    if (!matchedRows.length) return;
    headers = sh.getRange(1, 1, 1, c.cols).getDisplayValues()[0];

    matchedRows.forEach(x => {
      out.push(makeRecord_(c, x.rowNumber, x.shownRow, headers, x.dateSort));
    });
  });

  // 保留原本排序：倉別 → 品號 → 日期（舊到新）。
  out.sort((a,b)=>{
    if(a.sheetOrder!==b.sheetOrder)return a.sheetOrder-b.sheetOrder;
    const byItem=String(a.itemNo).localeCompare(String(b.itemNo),'zh-Hant',{numeric:true});
    if(byItem)return byItem;
    if(!a.dateSort&&!b.dateSort)return a.row-b.row;
    if(!a.dateSort)return 1;
    if(!b.dateSort)return -1;
    return a.dateSort-b.dateSort||a.row-b.row;
  });

  return {results:out,count:out.length};
}

/**
 * 入庫／出庫
 * 重點：在既有公式後面繼續加減，保留之前所有異動公式。
 */
function performStockMovement(p) {
  validatePayload_(p);

  const c = config_(p.sheetName);
  const sh = getSheet_(c);
  const row = validRow_(sh, p.row);

  if (p.action !== 'in' && p.action !== 'out') {
    throw new Error('異動類型錯誤。');
  }

  const allowedOperators = ['湯秀蓮','李政松','邱慶哲','黃文廷'];
  const operator = String(p.operator || '').trim();
  if (!allowedOperators.includes(operator)) {
    throw new Error('請先選擇有效的操作人員，再進行入庫或出庫。');
  }

  const inputs = [
    {
      key:'qty1',
      col:c.qty1Col,
      label:c.type === 'double' ? '數量1' : '數量'
    },
    {
      key:'qty2',
      col:c.qty2Col,
      label:'數量2'
    }
  ];

  const changes = [];

  inputs.forEach(x=>{
    if(
      !x.col ||
      p[x.key] === '' ||
      p[x.key] === null ||
      p[x.key] === undefined
    ) return;

    const amount = Number(p[x.key]);

    if(!Number.isFinite(amount)||amount<=0){
      throw new Error(x.label+'請輸入大於 0 的數字，或留空不異動。');
    }

    const cell = sh.getRange(row,x.col);
    const currentRaw = cell.getValue();
    const current = currentRaw === '' ? 0 : Number(currentRaw);

    if(!Number.isFinite(current)){
      throw new Error(x.label+'目前不是數字，請先檢查試算表。');
    }

    const next = p.action === 'in'
      ? current + amount
      : current - amount;

    if(next<0){
      throw new Error(x.label+'庫存不足，不能出庫。');
    }

    changes.push({
      cell:cell,
      current:current,
      next:next,
      label:x.label,
      amount:amount
    });
  });

  if(!changes.length){
    throw new Error('請至少輸入一個數量。留空欄位不會異動。');
  }

  // 累積公式，不覆蓋先前的異動紀錄。
  changes.forEach(x=>{
    const operator = p.action === 'in' ? '+' : '-';
    const oldFormula = x.cell.getFormula();
    let newFormula;

    if(oldFormula && oldFormula.startsWith('=')){
      // 例如 =100-1，下一次入庫 1 會變成 =100-1+1。
      newFormula = oldFormula + operator + x.amount;
    }else{
      // 原本是固定數字時，以該數字作為公式起點。
      newFormula = '=' + x.current + operator + x.amount;
    }

    x.cell.setFormula(newFormula);
  });

  const actionText = p.action === 'in' ? '入庫' : '出庫';
  const noteText = String(p.note || '').trim();

  // 每次異動都寫入原本備註欄，保留舊紀錄並追加操作人員、類型、數量與備註。
  // 範例：邱慶哲入庫｜5｜退料；未填備註時：邱慶哲入庫｜5
  const noteQuantities = changes.map(x=>x.amount).join('、');
  const movementLog = operator + actionText + '｜' + noteQuantities +
    (noteText ? '｜' + noteText : '');
  appendNote_(sh, row, c.noteCol, movementLog);

  // 異動日期與備註分開儲存。
  sh.getRange(row,c.movementCol).setValue(new Date());
  sh.getRange(row,c.movementCol).setNumberFormat('yyyy/MM/dd (ddd) HH:mm');

  SpreadsheetApp.flush();

  const details = changes.map(x=>
    x.label+'：'+x.current+
    (p.action==='in'?' + ':' - ')+
    x.amount+' = '+x.next
  ).join('、');

  return {
    success:true,
    message:actionText+'完成：'+details
  };
}

/**
 * 編輯詳細資料。
 * 品號、品名、數量、ERP、確認、單位數量、異動、差異等欄位禁止修改。
 */
function updateMaterialFields(p) {
  validatePayload_(p);

  const c = config_(p.sheetName);
  const sh = getSheet_(c);
  const row = validRow_(sh,p.row);

  if(!Array.isArray(p.fields)||!p.fields.length){
    throw new Error('沒有可儲存的修改欄位。');
  }

  const headers = sh.getRange(1,1,1,c.cols).getDisplayValues()[0];
  const protectedNames = ['品號','品名','數量','數量1','數量2','ERP','確認','單位數量','異動','差異'];
  const protectedCols = [c.itemCol,c.nameCol,c.qty1Col,c.qty2Col,c.movementCol].filter(Boolean);
  const changed = [];

  p.fields.forEach(f=>{
    const col = Number(f.col);
    if(!Number.isInteger(col)||col<1||col>c.cols){
      throw new Error('修改欄位編號不正確。');
    }

    const label = String(headers[col-1]||'')
      .replace(/[\s\r\n：:（）()]/g,'')
      .toLowerCase();

    if(
      protectedCols.includes(col) ||
      protectedNames.some(n=>
        label===n.toLowerCase() ||
        (['erp','確認','單位數量','差異','品號','品名','異動'].includes(n.toLowerCase()) &&
          label.includes(n.toLowerCase()))
      )
    ){
      throw new Error('欄位「'+(headers[col-1]||col)+'」禁止修改。');
    }

    const cell = sh.getRange(row,col);
    const oldValue = String(cell.getDisplayValue()||'');
    const newValue = String(f.value===null||f.value===undefined?'':f.value);

    if(oldValue!==newValue){
      cell.setValue(newValue);
      changed.push(String(headers[col-1]||('第'+col+'欄')));
    }
  });

  if(!changed.length){
    throw new Error('資料沒有變更，無需儲存。');
  }

  SpreadsheetApp.flush();
  return {success:true,message:'資料修改完成：'+changed.join('、')};
}

/**
 * 修改庫位：只更新庫位，不增加備註、不更動異動日期。
 */
function updateMaterialLocations(p) {
  validatePayload_(p);

  const c = config_(p.sheetName);
  const sh = getSheet_(c);
  const row = validRow_(sh,p.row);
  const changes = [];

  const inputs = [
    {key:'loc1',col:c.loc1Col,label:'庫位1'},
    {key:'loc2',col:c.loc2Col,label:'庫位2'}
  ];

  inputs.forEach(x=>{
    if(!x.col)return;
    if(p[x.key]===undefined||p[x.key]===null||String(p[x.key]).trim()==='')return;

    const cell = sh.getRange(row,x.col);
    const oldValue = String(cell.getDisplayValue()||'').trim();
    const newValue = String(p[x.key]).trim();

    if(oldValue!==newValue){
      cell.setValue(newValue);
      changes.push(x.label);
    }
  });

  if(!changes.length){
    throw new Error('庫位沒有變更，無需儲存。');
  }

  SpreadsheetApp.flush();
  return {success:true,message:'庫位修改完成：'+changes.join('、')};
}

function validatePayload_(p) {
  if(!p||!p.sheetName||!p.row){
    throw new Error('資料資訊不足，請重新查詢。');
  }
}
function config_(name) {
  const c = SHEET_CONFIG.find(x=>x.name===String(name));
  if(!c)throw new Error('不允許操作此分頁：'+name);
  return c;
}
function getSheet_(c) {
  const sh = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(c.name);
  if(!sh)throw new Error('找不到分頁：'+c.name+'。請檢查 SHEET_CONFIG 的分頁名稱。');
  return sh;
}
function validRow_(sh,n) {
  const row = Number(n);
  if(!Number.isInteger(row)||row<2||row>sh.getLastRow()){
    throw new Error('資料列已失效，請重新查詢。');
  }
  return row;
}
function at_(row,col) {
  if(!col)return '';
  const value = row[col-1];
  return value===null||value===undefined?'':String(value);
}
function timestamp_() {
  return Utilities.formatDate(new Date(),Session.getScriptTimeZone(),'yyyy/MM/dd HH:mm:ss');
}
function appendNote_(sh,row,col,text) {
  if(!col)return;
  const cell = sh.getRange(row,col);
  const oldText = String(cell.getDisplayValue()||'').trim();
  cell.setValue(oldText?oldText+'\n'+text:text);
}

/**
 * 只讀取指定分頁、指定列的物料資料
 * 異動後局部刷新使用，不重新搜尋全部分頁。
 */
function getMaterialRecord(sheetName, rowNumber) {
  const c = config_(sheetName);
  const sh = getSheet_(c);
  const row = validRow_(sh, rowNumber);

  const values = sh.getRange(row, 1, 1, c.cols).getValues()[0];
  const shown = sh.getRange(row, 1, 1, c.cols).getDisplayValues()[0];
  const headers = sh.getRange(1, 1, 1, c.cols).getDisplayValues()[0];

  const dateValue = c.dateCol ? values[c.dateCol - 1] : null;
  let dateSort = 0;

  if (dateValue instanceof Date && !isNaN(dateValue.getTime())) {
    dateSort = dateValue.getTime();
  } else if (dateValue) {
    dateSort = Date.parse(dateValue) || 0;
  }

  return {
    sheetName: c.name,
    sheetOrder: c.order,
    type: c.type,
    row: row,
    itemNo: at_(shown, c.itemCol),
    itemName: at_(shown, c.nameCol),
    batch: at_(shown, c.batchCol),
    dateText: c.dateCol ? at_(shown, c.dateCol) : '',
    dateSort: dateSort,
    loc1: at_(shown, c.loc1Col),
    qty1: at_(shown, c.qty1Col),
    loc2: c.qty2Col ? at_(shown, c.loc2Col) : '',
    qty2: c.qty2Col ? at_(shown, c.qty2Col) : '',
    headers: headers,
    allFields: shown.slice(0, c.cols)
  };
}
